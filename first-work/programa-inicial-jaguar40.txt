Programa com os registradores e endereços definidos:

1.  LOAD r0       x"000"	(carrega o valor da linha 1 da memoria de dados (1) no r0)
2.  LOAD r1       x"001"	(carrega o valor da linha 0 da memoria de dados (2) no r1)
3.  ADD  r2 r0 r1 		(soma r0 com r1 e guarda no r2):		"2+1 = 3"
4.  SUB  r3 r2 r1 		(subtrai r2 com r1 e guarda no r3): 		"3-2 = 1"
5.  AND  r4 r2 r1 		(faz uma AND do r2 com o r1 e guarda no r4)	"3 AND 2 = 2"
6.  OR   r5 r2 r1 		(faz uma OR do r2 com o r1 e guarda no r5)	"3 OR 2 = 3"
7.  BEQ     r2 r1     		(não vai passar no BEQ):			"3-2 != 0"
8.  BEQ     r0 r3 x"014"	(passa no BEQ e pula pra linha 20):		"1-1 = 0"	
20. ADD  r6 r5 r2		(soma r5 com r2 e guarda no r6):		"3+3 = 6"
21. STR     r6    x"004"	(guarda o valor do r6 (que é 6) na linha 5 da memoria de dados)
22. JMP	 	  x"01E"	(pula pra linha 30)  
30. HLT
----------------------------------------------------------------------------
